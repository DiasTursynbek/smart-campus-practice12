+import type { ExplainInput } from "./contract";

+// NOTE: a real model adapter must run server-side (behind your own backend endpoint)
+// so that API keys never reach the browser. Only this mock exists in the client.
+
+export interface LlmAdapter {
+  name: string;
+  generate(prompt: { system: string; user: string }, signal: AbortSignal): Promise<string>;
+}
+
+export type MockMode = "normal" | "slow" | "error" | "invalid";
+
+const SLOW_MS = 10_000;
+
+function wait(ms: number, signal: AbortSignal): Promise<void> {
+  return new Promise((resolve, reject) => {
+    if (signal.aborted) {
+      reject(new Error("Aborted"));
+      return;
+    }
+    const onAbort = () => {
+      clearTimeout(timer);
+      reject(new Error("Aborted"));
+    };
+    const timer = setTimeout(() => {
+      signal.removeEventListener("abort", onAbort);
+      resolve();
+    }, ms);
+    signal.addEventListener("abort", onAbort, { once: true });
+  });
+}
+
+// The mock only sees the prompt text, so it reads the input back from the JSON inside <data>.
+function readInput(prompt: { user: string }): ExplainInput {
+  const m = /^<data>([\s\S]*)<\/data>$/.exec(prompt.user);
+  if (!m) throw new Error("Malformed prompt");
+  return JSON.parse(m[1]) as ExplainInput;
+}
+
+function template(i: ExplainInput): string {
+  const label = i.roomLabel;
+  switch (i.reasonCode) {
+    case "AVAILABLE":
+      return `${label} is available from ${i.start} to ${i.end}. This is only a check, not a booking confirmation.`;
+    case "NO_ROOM":
+      return "No room is selected, so the request cannot be checked.";
+    case "INVALID_TIME":
+      return `${label} is not available because the start or end time is not valid.`;
+    case "INVALID_RANGE":
+      return `${label} is not available because the end time must be after the start time.`;
+    case "PAST_START":
+      return `${label} is not available because the start time ${i.start} is not after the current time.`;
+    case "TOO_LONG":
+      return `${label} is not available from ${i.start} to ${i.end} because bookings cannot exceed 120 minutes.`;
+    case "ROOM_BLOCKED":
+      return `${label} is blocked and cannot be booked from ${i.start} to ${i.end}.`;
+    case "OVERLAP":
+      return `${label} is not available from ${i.start} to ${i.end} because it overlaps an existing booking.`;
+  }
+}

+export function createMockAdapter(mode: MockMode, delayMs = 300): LlmAdapter {
+  return {
+    name: `mock-${mode}`,
+    async generate(prompt, signal) {

+        return template(readInput(prompt));

+      await wait(delayMs, signal);

+        // Deliberately contradicts the real status.
+        return input.isAvailable
+          ? "The room is blocked and not available for any time."
+          : "The room is available for any time you like.";
+      }
+      return template(input);
+    },
+  };

+import type { ReasonCode } from "@/types";
+
+export const MAX_ROOM_LABEL = 60;
+export const MAX_EXPLANATION = 280;
+
+export const REASON_CODES: readonly ReasonCode[] = [
+  "AVAILABLE",
+  "NO_ROOM",
+  "INVALID_TIME",
+  "INVALID_RANGE",
+  "PAST_START",
+  "TOO_LONG",
+  "ROOM_BLOCKED",
+  "OVERLAP",
+];
+
+export interface ExplainInput {
+  roomLabel: string;
+  start: string; // HH:MM
+  end: string; // HH:MM
+  isAvailable: boolean;
+  reasonCode: ReasonCode;
+}
+
+export type ValidationResult = { ok: true } | { ok: false; error: string };
+
+const HHMM = /^([01]\d|2[0-3]):[0-5]\d$/;
+// Any clock-like token inside generated text (used to detect invented times).
+const TIME_IN_TEXT = /\b\d{1,2}:\d{2}\b/g;
+
+export function validateExplainInput(input: ExplainInput): ValidationResult {
+  // Defensive: callers may pass data that does not match the static type.
+  const i = input as unknown as Record<string, unknown> | null;
+  if (typeof i !== "object" || i === null) {
+    return { ok: false, error: "Input must be an object." };
+  }
+  const { roomLabel, start, end, isAvailable, reasonCode } = i;
+
+  if (typeof roomLabel !== "string" || roomLabel.trim().length === 0) {
+    return { ok: false, error: "Room label must not be empty." };
+  }
+  if (roomLabel.length > MAX_ROOM_LABEL) {
+    return { ok: false, error: `Room label must be at most ${MAX_ROOM_LABEL} characters.` };
+  }
+  if (typeof start !== "string" || !HHMM.test(start)) {
+    return { ok: false, error: "Start must be in HH:MM format." };
+  }
+  if (typeof end !== "string" || !HHMM.test(end)) {
+    return { ok: false, error: "End must be in HH:MM format." };
+  }
+  if (typeof reasonCode !== "string" || !REASON_CODES.includes(reasonCode as ReasonCode)) {
+    return { ok: false, error: "Unknown reason code." };
+  }
+  if (typeof isAvailable !== "boolean") {
+    return { ok: false, error: "isAvailable must be a boolean." };
+  }
+  if (isAvailable !== (reasonCode === "AVAILABLE")) {
+    return { ok: false, error: "isAvailable does not agree with reasonCode." };
+  }
+  return { ok: true };
+}
+
+export function validateExplanation(text: string, input: ExplainInput): ValidationResult {
+  if (typeof text !== "string" || text.trim().length === 0) {
+    return { ok: false, error: "Explanation is empty." };
+  }
+  if (text.length > MAX_EXPLANATION) {
+    return { ok: false, error: `Explanation is longer than ${MAX_EXPLANATION} characters.` };
+  }
+
+  const sentences = text.split(/[.!?]+/).filter((s) => s.trim().length > 0);
+  if (sentences.length > 2) {
+    return { ok: false, error: "Explanation has more than two sentences." };
+  }
+
+  // Remove negated forms first so "not available" is not read as "available".
+  const lower = text.toLowerCase();
+  const claimsUnavailable = /\b(unavailable|not available|blocked)\b/.test(lower);
+  const claimsAvailable = /\bavailable\b/.test(
+    lower.replace(/\bnot available\b/g, " ").replace(/\bunavailable\b/g, " ")
+  );
+  if (!input.isAvailable && claimsAvailable) {
+    return { ok: false, error: "Explanation says available but the room is unavailable." };
+  }
+  if (input.isAvailable && claimsUnavailable) {
+    return { ok: false, error: "Explanation says unavailable but the room is available." };
+  }
+
+  const times = text.match(TIME_IN_TEXT) ?? [];
+  for (const t of times) {
+    if (t !== input.start && t !== input.end) {
+      return { ok: false, error: `Explanation mentions an unsupplied time (${t}).` };
+    }
+  }
+  return { ok: true };
+}

+import type { ReasonCode } from "@/types";
+import type { LlmAdapter } from "./adapters";
+import { ExplainInput, validateExplainInput, validateExplanation } from "./contract";
+import { buildPrompt } from "./prompt";
+
+export interface ExplainResult {
+  source: "model" | "fallback";
+  explanation: string;
+  error?: string;
+}
+
+const FALLBACKS: Record<ReasonCode, string> = {
+  AVAILABLE: "The room is available for the requested time.",
+  NO_ROOM: "No room is selected, so the request cannot be checked.",
+  INVALID_TIME: "The room is not available because the start or end time is not valid.",
+  INVALID_RANGE: "The room is not available because the end time must be after the start time.",
+  PAST_START: "The room is not available because the start time must be after the current time.",
+  TOO_LONG: "The room is not available because bookings cannot exceed 120 minutes.",
+  ROOM_BLOCKED: "The room is blocked and cannot be booked.",
+  OVERLAP: "The room is not available because it overlaps an existing booking.",
+};
+
+const GENERIC_FALLBACK = "No explanation could be generated for this request.";
+
+export function fallbackExplanation(code: unknown): string {
+  return typeof code === "string" && Object.prototype.hasOwnProperty.call(FALLBACKS, code)
+    ? FALLBACKS[code as ReasonCode]
+    : GENERIC_FALLBACK;
+}
+
+export async function explainAvailability(
+  input: ExplainInput,
+  adapter: LlmAdapter,
+  { timeoutMs = 3000 }: { timeoutMs?: number } = {}
+): Promise<ExplainResult> {
+  const check = validateExplainInput(input);
+  if (!check.ok) {
+    return {
+      source: "fallback",
+      explanation: fallbackExplanation((input as Partial<ExplainInput> | null)?.reasonCode),
+      error: `Invalid input: ${check.error}`,
+    };
+  }
+
+  // Work on a private copy so the caller's object is never touched.
+  const snapshot: ExplainInput = { ...input };
+  const fallback = (error: string): ExplainResult => ({
+    source: "fallback",
+    explanation: fallbackExplanation(snapshot.reasonCode),
+    error,
+  });
+
+  const controller = new AbortController();
+  let timer: ReturnType<typeof setTimeout> | undefined;
+  const timeout = new Promise<never>((_, reject) => {
+    timer = setTimeout(() => {
+      controller.abort();
+      reject(new Error(`Model timed out after ${timeoutMs} ms`));
+    }, timeoutMs);
+  });
+
+  let text: string;
+  try {
+    const pending = adapter.generate(buildPrompt(snapshot), controller.signal);
+    pending.catch(() => undefined); // avoid unhandled rejection if the timeout wins
+    text = await Promise.race([pending, timeout]);
+  } catch (e) {
+    return fallback(e instanceof Error ? e.message : "Model call failed");
+  } finally {
+    clearTimeout(timer);
+  }
+
+  const out = validateExplanation(text, snapshot);
+  if (!out.ok) return fallback(`Invalid model output: ${out.error}`);
+  return { source: "model", explanation: text.trim() };
+}
